# 把輸入設為任意合法參數組合；輸出 balances_charted 必須逐項完全等於 balances_raw；依 PRD-12 圖表餘額不得進行任何下限截斷，計算核心須忠實反映真實餘額。
def test_mr_balances_charted_equals_raw(base_params: Params) -> None:
    res = calculate(base_params)
    assert res.balances_charted == res.balances_raw


# 把每月支出減少 X 並將固定年支出增加 12X；輸出 target_fund 與餘額軌跡必須完全相同；依 PRD-08 兩者合併計算且共用同一組鎖定現齡的通膨基準。
def test_mr_monthly_and_annual_expense_fungibility(base_params: Params) -> None:
    shift_monthly = 5_000.0
    p_shifted = replace(
        base_params,
        monthly_expense_today=base_params.monthly_expense_today - shift_monthly,
        annual_recurring_expense=base_params.annual_recurring_expense + shift_monthly * 12,
    )
    res_base = calculate(base_params)
    res_shifted = calculate(p_shifted)

    assert res_shifted.target_fund == pytest.approx(res_base.target_fund, rel=1e-7)
    assert res_shifted.balances_raw == pytest.approx(res_base.balances_raw, rel=1e-7)


# 把大筆支出分別設定在退休前（Ae < Ar）與壽命後（Ae > Ad）；輸出各項指標必須與未加入前完全一致；依 PRD-05 僅採計提領期間內支出，區間外項目皆被排除。
def test_mr_out_of_range_lump_sum_invariance(base_params: Params) -> None:
    p_with_irrelevant_lumps = replace(
        base_params,
        lump_sums=(
            LumpSum(age=base_params.current_age, amount=300_000.0),
            LumpSum(age=base_params.retirement_age - 1, amount=400_000.0),
            LumpSum(age=base_params.life_expectancy + 1, amount=500_000.0),
        ),
    )
    res_base = calculate(base_params)
    res_filtered = calculate(p_with_irrelevant_lumps)

    assert res_filtered.target_fund == pytest.approx(res_base.target_fund, rel=1e-7)
    assert res_filtered.projected_savings == pytest.approx(res_base.projected_savings, rel=1e-7)
    assert res_filtered.retirement_gap == pytest.approx(res_base.retirement_gap, rel=1e-7)
    assert res_filtered.balances_raw == pytest.approx(res_base.balances_raw, rel=1e-7)


# 把退休期間內多筆大筆支出的輸入順序打亂對調；輸出 target_fund 與餘額軌跡必須完全相同；因大筆支出按發生年齡各自折算計入，輸入順序不影響計算結果。
def test_mr_lump_sum_order_invariance(base_params: Params) -> None:
    lump_a = LumpSum(age=65, amount=200_000.0)
    lump_b = LumpSum(age=72, amount=500_000.0)
    p_order1 = replace(base_params, lump_sums=(lump_a, lump_b))
    p_order2 = replace(base_params, lump_sums=(lump_b, lump_a))

    res1 = calculate(p_order1)
    res2 = calculate(p_order2)

    assert res2.target_fund == pytest.approx(res1.target_fund, rel=1e-7)
    assert res2.balances_raw == pytest.approx(res1.balances_raw, rel=1e-7)


# 把同一年齡發生的兩筆大筆支出合併為該年的一筆總額；輸出 target_fund 與提領軌跡必須完全相同；依 PRD-05 同一年齡之支出共用相同通膨與折現因子，具加法可合併性。
def test_mr_lump_sum_same_age_additivity(base_params: Params) -> None:
    p_split = replace(
        base_params,
        lump_sums=(
            LumpSum(age=68, amount=250_000.0),
            LumpSum(age=68, amount=350_000.0),
        ),
    )
    p_merged = replace(
        base_params,
        lump_sums=(LumpSum(age=68, amount=600_000.0),),
    )
    res_split = calculate(p_split)
    res_merged = calculate(p_merged)

    assert res_merged.target_fund == pytest.approx(res_split.target_fund, rel=1e-7)
    assert res_merged.balances_raw == pytest.approx(res_split.balances_raw, rel=1e-7)


# 把現有資產與每月投資同時等比例放大 k 倍；輸出 projected_savings 必須剛好放大 k 倍；依 PRD-01 累積期名目利率複利公式與期末年金對資金規模皆具線性齐次性。

    p_scaled = replace(

# 把現有資產與每月投資分別單獨投入計算累積值；兩次輸出的 projected_savings 相加必須等於兩者同時投入時的結果；依 PRD-01 複利終值公式滿足線性疊加原理。

# 在退休前報酬率為 0 時，將月存金額由 M 變更為 2M；兩次結果的差值 (res_2M - res_M) 必須等於 (res_M - res_0M)；依 PRD-01 在 r=0 時累積資產退化為純線性累加。
def test_mr_zero_return_projected_savings_linearity(base_params: Params) -> None:
    m = 15_000.0
    p_0m = replace(base_params, pre_retirement_return=0.0, monthly_investment=0.0)
    p_1m = replace(base_params, pre_retirement_return=0.0, monthly_investment=m)
    p_2m = replace(base_params, pre_retirement_return=0.0, monthly_investment=2 * m)

    res_0m = calculate(p_0m)
    res_1m = calculate(p_1m)
    res_2m = calculate(p_2m)

    diff1 = res_1m.projected_savings - res_0m.projected_savings
    diff2 = res_2m.projected_savings - res_1m.projected_savings

    assert diff2 == pytest.approx(diff1, rel=1e-7)
