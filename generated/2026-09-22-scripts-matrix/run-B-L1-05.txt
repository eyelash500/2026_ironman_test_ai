# 1. 基礎契約與型別測試 (Contract & Data Integrity)
# ==============================================================================


def test_calculate_returns_result_instance(base_params: Params):
    """驗證 calculate 執行成功並回傳 Result 實例。"""
    res = calculate(base_params)
    assert isinstance(res, Result), "calculate 應回傳 Result 形態"


def test_result_scalar_fields_are_finite_floats(base_params: Params):
    """驗證回傳的純量數值欄位型別為 float 且皆為有限數值（非 NaN、非 Inf）。"""
    res = calculate(base_params)

    for field_name in ("projected_savings", "target_fund", "retirement_gap"):
        val = getattr(res, field_name)
        assert isinstance(val, (int, float)), f"{field_name} 必須是浮點數或整數"
        assert math.isfinite(float(val)), f"{field_name} 不得為 NaN 或 Inf"


def test_result_balances_series_types_and_length(base_params: Params):
    """驗證 balances_raw 與 balances_charted 皆為 tuple，內容為有限浮點數，且長度與年齡跨度一致。"""
    res = calculate(base_params)

    assert isinstance(res.balances_raw, tuple), "balances_raw 必須是 tuple"
    assert isinstance(res.balances_charted, tuple), "balances_charted 必須是 tuple"

    # balances 長度應涵蓋當前年齡至預期壽命區間
    expected_span = base_params.life_expectancy - base_params.current_age
    assert len(res.balances_raw) in (expected_span, expected_span + 1), (
        f"balances_raw 長度應為 {expected_span} 或 {expected_span + 1}，實際為 {len(res.balances_raw)}"
    )
    assert len(res.balances_raw) == len(res.balances_charted), (
        "balances_raw 與 balances_charted 之序列長度應完全一致"
    )

    for i, val in enumerate(res.balances_raw):
        assert isinstance(val, (int, float)), f"balances_raw[{i}] 必須為數值型態"
        assert math.isfinite(float(val)), f"balances_raw[{i}] 不得為 NaN 或 Inf"

    for i, val in enumerate(res.balances_charted):
        assert isinstance(val, (int, float)), f"balances_charted[{i}] 必須為數值型態"
        assert math.isfinite(float(val)), f"balances_charted[{i}] 不得為 NaN 或 Inf"

def test_projected_savings_monotonically_increases_with_current_savings(base_params: Params):
    """初始資產增加時，退休時累積資產 projected_savings 應嚴格遞增。"""
    p_high = replace(base_params, current_savings=base_params.current_savings + 200_000.0)

    res_base = calculate(base_params)
    res_high = calculate(p_high)

    assert res_high.projected_savings > res_base.projected_savings, (
        "較高的 current_savings 應產生較高的 projected_savings"

def test_positive_pre_retirement_lump_sum_increases_projected_savings(base_params: Params):
    """退休前獲得單筆資金（如繼承或分紅），應使退休時累積資產增加。"""
    lump = LumpSum(age=base_params.current_age + 10, amount=300_000.0)
    p_with_lump = replace(base_params, lump_sums=(lump,))

    res_base = calculate(base_params)
    res_lump = calculate(p_with_lump)

    assert res_lump.projected_savings > res_base.projected_savings, (
        "退休前有單筆正向現金流入時，projected_savings 應增加"
    )


def test_multiple_lump_sums_handling(base_params: Params):
    """測試傳入多筆不同年齡之單筆收支，函式可正常運算不拋例外。"""
    lumps = (
        LumpSum(age=base_params.current_age + 5, amount=100_000.0),
        LumpSum(age=base_params.retirement_age + 5, amount=-150_000.0),
        LumpSum(age=base_params.retirement_age + 10, amount=50_000.0),
    )
    p_lumps = replace(base_params, lump_sums=lumps)
    res = calculate(p_lumps)

    assert isinstance(res, Result)
    assert math.isfinite(res.projected_savings)
    assert math.isfinite(res.target_fund)


def test_pension_and_other_incomes_execution(base_params: Params):
    """測試包含勞保年金、勞退月退與其他收入的完整計算情境。"""

        base_params,

    assert math.isfinite(res.retirement_gap)

def test_exact_annuity_compounding_convention(base_params: Params):
    """驗證精確的複利年金累積公式。"""
    # 僅留月投資額，初始資產為 0，零利率環境
