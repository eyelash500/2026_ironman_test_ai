def test_calculate_returns_result_instance(default_params: Params):
    """驗證 calculate 執行後返回 Result 型別物件且為 dataclass。"""
    res = calculate(default_params)
    assert isinstance(res, Result)
    assert is_dataclass(res)


def test_result_fields_exist_and_types(default_params: Params):
    """驗證 Result 各屬性存在且型別符合定義。"""
    res = calculate(default_params)

    assert isinstance(res.projected_savings, (int, float))
    assert isinstance(res.target_fund, (int, float))
    assert isinstance(res.retirement_gap, (int, float))
    assert isinstance(res.balances_raw, tuple)
    assert isinstance(res.balances_charted, tuple)


def test_result_numeric_values_are_finite(default_params: Params):
    """驗證計算出的數值為有限實數（非 NaN 且非 Inf）。"""
    res = calculate(default_params)

    for field_name in ("projected_savings", "target_fund", "retirement_gap"):
        val = getattr(res, field_name)
        assert not math.isnan(val), f"{field_name} should not be NaN"
        assert not math.isinf(val), f"{field_name} should not be infinite"

    for idx, val in enumerate(res.balances_raw):
        assert not math.isnan(val), f"balances_raw[{idx}] should not be NaN"
        assert not math.isinf(val), f"balances_raw[{idx}] should not be infinite"

    for idx, val in enumerate(res.balances_charted):
        assert not math.isnan(val), f"balances_charted[{idx}] should not be NaN"
        assert not math.isinf(val), f"balances_charted[{idx}] should not be infinite"


def test_balances_tuple_elements_and_lengths(default_params: Params):
    """驗證 balances_raw 與 balances_charted 的長度與內容型別一致。"""
    res = calculate(default_params)

    assert len(res.balances_raw) == len(res.balances_charted)
    assert len(res.balances_raw) > 0
    assert all(isinstance(x, (int, float)) for x in res.balances_raw)
    assert all(isinstance(x, (int, float)) for x in res.balances_charted)


def test_balances_length_corresponds_to_age_span(default_params: Params):
    """驗證資產時序長度應與年齡跨度（起算至預期壽命）對應。"""
    res = calculate(default_params)
    span = default_params.life_expectancy - default_params.current_age
    # 常見實作長度為 span (半開區間) 或 span + 1 (閉區間)
    assert len(res.balances_raw) in (span, span + 1)

def test_monotonicity_higher_savings_increases_projected_savings(default_params: Params):
    """不變量：初始資產越高，退休時預估累積資產應越高。"""

        current_age=p_low.current_age,

def test_lump_sum_positive_inflow_increases_savings(default_params: Params):
    """退休前有單筆資金流入（如繼承或年終獎金），應提升退休預估資產。"""
    p_base = default_params
    p_with_lump = Params(
        current_age=p_base.current_age,
        retirement_age=p_base.retirement_age,
        life_expectancy=p_base.life_expectancy,
        current_savings=p_base.current_savings,
        monthly_investment=p_base.monthly_investment,
        monthly_expense_today=p_base.monthly_expense_today,
        lump_sums=(LumpSum(age=40, amount=500_000.0),),
    )

    res_base = calculate(p_base)
    res_with_lump = calculate(p_with_lump)

    assert res_with_lump.projected_savings > res_base.projected_savings


def test_lump_sum_negative_outflow_decreases_savings(default_params: Params):
    """退休前有單筆大額支出（負金額），應降低退休預估資產。"""
    p_base = default_params
    p_with_expense = Params(
        current_age=p_base.current_age,
        retirement_age=p_base.retirement_age,
        life_expectancy=p_base.life_expectancy,
        current_savings=p_base.current_savings,
        monthly_investment=p_base.monthly_investment,
        monthly_expense_today=p_base.monthly_expense_today,
        lump_sums=(LumpSum(age=45, amount=-300_000.0),),
    )

    res_base = calculate(p_base)
    res_with_expense = calculate(p_with_expense)

    assert res_with_expense.projected_savings < res_base.projected_savings
