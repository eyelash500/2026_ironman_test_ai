import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "shadow"))

from calc_fixed import LumpSum, Params, Result, calculate  # noqa: E402


def test_result_structure():
    """Verify that calculate returns a correctly typed Result object."""
    p = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=100_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=50_000.0,
        annual_recurring_expense=10_000.0,
        labor_insurance_pension=20_000.0,
        labor_insurance_start_age=65,
        labor_pension_monthly=5_000.0,
        labor_pension_start_age=65,
        other_income=2_000.0,
        pre_retirement_return=0.06,
        post_retirement_return=0.04,
        inflation_rate=0.02,
        lump_sums=(LumpSum(age=40, amount=500_000.0),)
    )
    res = calculate(p)
    assert isinstance(res, Result)
    assert isinstance(res.projected_savings, float)
    assert isinstance(res.target_fund, float)
    assert isinstance(res.retirement_gap, float)
    assert isinstance(res.balances_raw, tuple)
    assert isinstance(res.balances_charted, tuple)


def test_zero_scenario():
    """When there is no money involved, all monetary results should be 0.0."""
    p = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=0.0,
        annual_recurring_expense=0.0,
        pre_retirement_return=0.0,
        post_retirement_return=0.0,
        inflation_rate=0.0
    )
    res = calculate(p)
    assert res.projected_savings == 0.0
    assert res.target_fund == 0.0
    assert res.retirement_gap == 0.0
    for b in res.balances_raw:
        assert b == 0.0
    for b in res.balances_charted:
        assert b == 0.0
