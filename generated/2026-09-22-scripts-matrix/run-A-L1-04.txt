import sys

ROOT = Path(__file__).resolve().parent.parent

from calc_fixed import LumpSum, Params, Result, calculate  # noqa: E402


def test_calculate_returns_expected_types():
    params = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=90,
        current_savings=100000.0,
        monthly_investment=10000.0,
        monthly_expense_today=50000.0
    )
    result = calculate(params)
    
    assert isinstance(result, Result)
    assert isinstance(result.projected_savings, float)
    assert isinstance(result.target_fund, float)
    assert isinstance(result.retirement_gap, float)
    assert isinstance(result.balances_raw, tuple)
    assert isinstance(result.balances_charted, tuple)

def test_higher_monthly_investment_increases_projected_savings():
    base_params = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=90,
        current_savings=100000.0,
        monthly_investment=10000.0,
        monthly_expense_today=50000.0,
        pre_retirement_return=0.05
    )
    high_invest_params = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=90,
        current_savings=100000.0,
        monthly_investment=20000.0,  # 增加投資額
        monthly_expense_today=50000.0,
        pre_retirement_return=0.05
    )
    
    res_base = calculate(base_params)
    res_high = calculate(high_invest_params)
    
    assert res_high.projected_savings > res_base.projected_savings


def test_higher_expenses_increases_target_fund():

def test_pre_retirement_lump_sum_increases_projected_savings():
    base_params = Params(
        current_age=30,
        retirement_age=60,
        life_expectancy=85,
        current_savings=0.0,
        monthly_investment=10000.0,
        monthly_expense_today=50000.0,
        pre_retirement_return=0.05
    )
    lump_params = Params(
        current_age=30,
        retirement_age=60,
        life_expectancy=85,
        current_savings=0.0,
        monthly_investment=10000.0,
        monthly_expense_today=50000.0,
        pre_retirement_return=0.05,
        lump_sums=(LumpSum(age=40, amount=2000000.0),)  # 退休前有一筆大額收入
    )
    
    res_base = calculate(base_params)
    res_lump = calculate(lump_params)
    
    assert res_lump.projected_savings > res_base.projected_savings
