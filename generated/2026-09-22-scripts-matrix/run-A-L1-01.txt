def test_params_dataclass_and_defaults():
    """測試 Params 的建立以及預設值是否符合介面定義"""
    p = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=100000.0,
        monthly_investment=15000.0,
        monthly_expense_today=40000.0,
    )
    assert p.current_age == 30
    assert p.annual_recurring_expense == 0.0
    assert p.labor_insurance_pension == 0.0
    assert p.labor_insurance_start_age == 0
    assert p.labor_pension_monthly == 0.0
    assert p.labor_pension_start_age == 0
    assert p.other_income == 0.0
    assert p.pre_retirement_return == 0.08
    assert p.post_retirement_return == 0.04
    assert p.inflation_rate == 0.02
    assert p.lump_sums == ()


def test_result_dataclass():
    """測試 Result 資料結構建立"""
    r = Result(
        projected_savings=100.0,
        target_fund=200.0,
        retirement_gap=100.0,
        balances_raw=(100.0, 150.0),
        balances_charted=(100.0, 150.0)
    )
    assert r.projected_savings == 100.0
    assert r.balances_raw == (100.0, 150.0)
