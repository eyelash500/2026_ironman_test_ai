# 輔助計算函式（嚴格依據 PRD-01 ~ PRD-10 規格建立獨立驗證模型）
# ---------------------------------------------------------------------------
def _compute_expected_projected_savings(p: Params) -> float:
    """依 PRD-01 計算累積期預估儲蓄。"""
    n_months = (p.retirement_age - p.current_age) * 12
    if p.pre_retirement_return == 0.0:
        return p.current_savings + p.monthly_investment * n_months

    r_m = p.pre_retirement_return / 12.0
    fv_savings = p.current_savings * ((1.0 + r_m) ** n_months)
    fv_investment = p.monthly_investment * (((1.0 + r_m) ** n_months - 1.0) / r_m)
    return fv_savings + fv_investment


def _compute_expected_nominal_net_expenses(p: Params) -> list[float]:
    """依 PRD-02, 05, 06, 07, 08, 09, 10 計算退休各年度的名目淨支出。"""
    net_expenses = []
    base_annual_expense = p.monthly_expense_today * 12.0 + p.annual_recurring_expense

    for age in range(p.retirement_age, p.life_expectancy + 1):
        # PRD-02 & PRD-08: 基礎支出，通膨基準鎖現齡 A_c
        nom_expense = base_annual_expense * ((1.0 + p.inflation_rate) ** (age - p.current_age))

        # PRD-05: 大筆支出，僅 A_r <= A_e <= A_d 計入，通膨鎖 A_c
        for ls in p.lump_sums:
            if ls.age == age and (p.retirement_age <= ls.age <= p.life_expectancy):
                nom_expense += ls.amount * ((1.0 + p.inflation_rate) ** (age - p.current_age))

        # PRD-06 & PRD-10: 勞保年金，t >= A_p 時計入，通膨鎖 A_p
        nom_income = 0.0
        if p.labor_insurance_pension > 0.0 and age >= p.labor_insurance_start_age:
            nom_income += (p.labor_insurance_pension * 12.0) * (
                (1.0 + p.inflation_rate) ** (age - p.labor_insurance_start_age)
            )

        # PRD-07 & PRD-10: 勞退月領，t >= A_p 時計入，通膨鎖 A_p
        if p.labor_pension_monthly > 0.0 and age >= p.labor_pension_start_age:
            nom_income += (p.labor_pension_monthly * 12.0) * (
                (1.0 + p.inflation_rate) ** (age - p.labor_pension_start_age)
            )

        # PRD-10: 其他收入，通膨鎖 A_c（若有輸入）
        if p.other_income > 0.0:
            nom_income += (p.other_income * 12.0) * (
                (1.0 + p.inflation_rate) ** (age - p.current_age)
            )

        # PRD-09: 淨支出下限 max(0, 支出 - 收入)，盈餘不滾存
        net_expenses.append(max(0.0, nom_expense - nom_income))

    return net_expenses


def _compute_expected_target_fund(p: Params) -> float:
    """依 PRD-02 計算期初年金折現目標金額。"""
    net_expenses = _compute_expected_nominal_net_expenses(p)
    target = 0.0
    for k, net_exp in enumerate(net_expenses):
        target += net_exp / ((1.0 + p.post_retirement_return) ** k)
    return target

            current_age=30,
            retirement_age=60,

            monthly_expense_today=30_000.0,

            current_age=30,

    def test_pension_inflation_locked_to_start_age_not_current_age(self):
        """PRD-10: 勞保年金通膨基準鎖定請領年齡 A_p，第 t 年為 月額 x 12 x (1+i)^(t - A_p)。"""
        # 設計單年對照（退休只有 65 歲當年）：
        # t = 65, A_p = 65，因此在 65 歲時，勞保通膨次方為 (65 - 65) = 0，名目額為 20000 * 12 = 240,000。
        # 若錯誤鎖定 A_c=30，名目額會膨脹為 20000 * 12 * (1.03^35) = 675,326。
        p = Params(
            current_age=30,

            current_savings=0.0,
            monthly_investment=0.0,
            monthly_expense_today=30_000.0,
            labor_insurance_pension=20_000.0,
            labor_insurance_start_age=65,
            inflation_rate=0.03,
            post_retirement_return=0.04,
        )
        res = calculate(p)
        # 65 歲名目支出: 30000 * 12 * (1.03 ^ 35)
        expected_nom_expense = 360_000.0 * (1.03 ** 35)
        # 65 歲勞保名目收入: 20000 * 12 * (1.03 ^ 0) = 240,000
        expected_nom_income = 240_000.0
        expected_target = expected_nom_expense - expected_nom_income
        assert res.target_fund == pytest.approx(expected_target, rel=1e-4)

        """PRD-03 & PRD-12: 真實餘額為負時照實輸出負值，禁止截斷為 0。"""
        # 零存款、零投資、高支出，退休後餘額必然轉負

        assert any(b < 0.0 for b in res.balances_raw)
        # balances_charted 亦不得將負值截斷為 0
        assert any(b < 0.0 for b in res.balances_charted)
        assert res.balances_charted == res.balances_raw

    def test_exact_target_fund_depletes_at_life_expectancy(self):
        """期初年金數學性質: 當初始本金剛好等於 target_fund 時，終期餘額花完接近 0。"""
        p_base = Params(
            current_age=30,
            retirement_age=60,
            life_expectancy=75,
            current_savings=0.0,
            monthly_investment=0.0,
            monthly_expense_today=40_000.0,
            inflation_rate=0.02,
            post_retirement_return=0.04,
        )
        res_base = calculate(p_base)
        target = res_base.target_fund

        # 構造剛好等於 target_fund 的退休儲蓄（無退休前月存，純初始金額月增值至 target）
        # 或者直接設定 post-retirement 在 target_fund 下的提領表現
        # 這裡驗證最後一年的真實餘額在剛好儲備時趨近於 0
        n_months = (p_base.retirement_age - p_base.current_age) * 12
        r_m = p_base.pre_retirement_return / 12.0
        needed_savings = target / ((1.0 + r_m) ** n_months)

        p_funded = Params(
            current_age=p_base.current_age,
            retirement_age=p_base.retirement_age,
            life_expectancy=p_base.life_expectancy,
            current_savings=needed_savings,
            monthly_investment=0.0,
            monthly_expense_today=p_base.monthly_expense_today,
            inflation_rate=p_base.inflation_rate,
            post_retirement_return=p_base.post_retirement_return,
            pre_retirement_return=p_base.pre_retirement_return,
        )
        res_funded = calculate(p_funded)
        assert res_funded.projected_savings == pytest.approx(target, rel=1e-4)
        # 由於壽命終了時剛好花完，最後一年年底（或扣款後）餘額應貼近 0
        assert res_funded.balances_raw[-1] == pytest.approx(0.0, abs=100.0)

# 結果物件與退休缺口 (Retirement Gap) 整合測試
# ---------------------------------------------------------------------------
class TestResultIntegrity:
    def test_result_structure_and_types(self):
        """驗證 Result 各欄位形態與完整性。"""
        p = Params(
            current_age=30,
            retirement_age=60,
            life_expectancy=80,
            current_savings=1_000_000.0,
            monthly_investment=20_000.0,
            monthly_expense_today=40_000.0,
        )
        res = calculate(p)
        assert isinstance(res, Result)
        assert isinstance(res.projected_savings, float)
        assert isinstance(res.target_fund, float)
        assert isinstance(res.retirement_gap, float)
        assert isinstance(res.balances_raw, tuple)
        assert isinstance(res.balances_charted, tuple)

    def test_retirement_gap_consistency(self):
        """驗證 retirement_gap 與 target_fund、projected_savings 之間的關係。"""
        p = Params(
            current_age=30,
            retirement_age=60,
            life_expectancy=80,
            current_savings=500_000.0,
            monthly_investment=10_000.0,
            monthly_expense_today=50_000.0,
        )
        res = calculate(p)
        # retirement_gap 必然為 target_fund - projected_savings 或 projected_savings - target_fund
        diff = res.target_fund - res.projected_savings
        assert (res.retirement_gap == pytest.approx(diff, rel=1e-5)) or (
            res.retirement_gap == pytest.approx(-diff, rel=1e-5)
        )
