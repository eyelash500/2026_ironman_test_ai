import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "shadow"))

from calc_fixed import LumpSum, Params, Result, calculate  # noqa: E402


def test_prd_01_accumulation():
    """PRD-01: Accumulation phase with normal return rate."""
    p_normal = Params(
        current_age=30,
        retirement_age=60,
        life_expectancy=90,
        current_savings=100_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=0.0,
        pre_retirement_return=0.06
    )
    res = calculate(p_normal)
    # n = 360 months, monthly_r = 0.005
    # FV of current savings + FV of ordinary annuity (end-of-month)
    expected_savings = 100_000.0 * (1.005 ** 360) + 10_000.0 * ((1.005 ** 360) - 1) / 0.005
    assert res.projected_savings == pytest.approx(expected_savings, rel=1e-5)

def test_prd_02_08_target_fund_and_recurring_expense():
    """PRD-02, 08: Target fund discounting and inflation with monthly & annual expenses."""
    p = Params(
        current_age=50,
        retirement_age=60,
        life_expectancy=62, # 60, 61, 62
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=10_000.0,
        annual_recurring_expense=50_000.0,
        pre_retirement_return=0.0,
        post_retirement_return=0.04,
        inflation_rate=0.02
    )
    res = calculate(p)
    # Annual expense at current age = 10,000 * 12 + 50,000 = 170,000
    # Inflation anchored at A_c (50)
    c_60 = 170_000.0 * (1.02 ** 10)
    c_61 = 170_000.0 * (1.02 ** 11)
    c_62 = 170_000.0 * (1.02 ** 12)
    # Beginning of year discounting back to A_r (60)
    expected_target = c_60 + c_61 / 1.04 + c_62 / (1.04 ** 2)
    
    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)
    assert res.retirement_gap == pytest.approx(res.target_fund - res.projected_savings, rel=1e-5)


def test_prd_04_input_validation():
    """PRD-04: Age validation and non-negative amounts must raise an exception."""
    # A_c >= A_r
    with pytest.raises(Exception):
        calculate(Params(current_age=60, retirement_age=60, life_expectancy=80, current_savings=0, monthly_investment=0, monthly_expense_today=0))
    # A_r >= A_d
    with pytest.raises(Exception):
        calculate(Params(current_age=30, retirement_age=60, life_expectancy=60, current_savings=0, monthly_investment=0, monthly_expense_today=0))
    # Negative amounts
    with pytest.raises(Exception):
        calculate(Params(current_age=30, retirement_age=60, life_expectancy=80, current_savings=-100, monthly_investment=0, monthly_expense_today=0))


def test_prd_05_lump_sums():
    """PRD-05: Lump sums outside bounds are ignored, inside are inflated from A_c."""
    p = Params(
        current_age=50,
        retirement_age=60,
        life_expectancy=62,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=0.0,
        pre_retirement_return=0.0,
        post_retirement_return=0.04,
        inflation_rate=0.02,
        lump_sums=(
            LumpSum(age=55, amount=1000), # Before retirement -> Ignore
            LumpSum(age=61, amount=2000), # Valid
            LumpSum(age=65, amount=3000), # After life expectancy -> Ignore
        )
    )
    res = calculate(p)
    # Only expense at age 61 is valid. Base = 2000, inflated from age 50
    c_61 = 2000.0 * (1.02 ** 11)
    expected_target = c_61 / 1.04
    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)


def test_prd_06_07_10_incomes():
    """PRD-06, 07, 10: Incomes deduct from expense, nominal at A_p."""
    p = Params(
        current_age=50,
        retirement_age=60,
        life_expectancy=61,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=20_000.0,  # 240,000/yr base
        labor_insurance_pension=5_000.0, # 60,000/yr base
        labor_insurance_start_age=61,    # Starts at 61
        labor_pension_monthly=0.0,
        pre_retirement_return=0.0,
        post_retirement_return=0.04,
        inflation_rate=0.02
    )
    res = calculate(p)
    
    # Age 60: exp = 240,000 * 1.02^10. inc = 0
    c_60 = 240_000.0 * (1.02 ** 10)
    
    # Age 61: exp = 240,000 * 1.02^11. inc = 60,000 * 1.02^0 (starts at 61)
    c_61 = 240_000.0 * (1.02 ** 11) - 60_000.0
    
    expected_target = c_60 + c_61 / 1.04
    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)


def test_prd_09_expense_floor():
    """PRD-09: Expense lower bound floored at 0."""
    p = Params(
        current_age=50,
        retirement_age=60,
        life_expectancy=61,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=10_000.0,  # 120,000/yr base
        labor_insurance_pension=20_000.0, # 240,000/yr base
        labor_insurance_start_age=60,
        pre_retirement_return=0.0,
        post_retirement_return=0.04,
        inflation_rate=0.02
    )
    res = calculate(p)
    # Income (240k inflated from 60) consistently exceeds expense (120k inflated from 50)
    # Target fund must be exactly 0 due to max(0, expense - income) floor
    assert res.target_fund == 0.0


def test_prd_03_12_balances():
    """PRD-03, 12: balances_charted matches balances_raw, negative allowed."""
    p = Params(
        current_age=50,
        retirement_age=60,
        life_expectancy=62,
        current_savings=1_000_000.0,
        monthly_investment=0.0,
        monthly_expense_today=100_000.0, # Base 1,200,000/yr
        pre_retirement_return=0.0,
        post_retirement_return=0.04,
        inflation_rate=0.02
    )
    res = calculate(p)
    
    assert res.balances_charted == res.balances_raw
    assert len(res.balances_raw) == 62 - 60 + 1
    # Savings (1M) is immediately depleted by high expenses (>1.46M), must yield negative values
    assert any(b < 0 for b in res.balances_raw)
