def make_params(**overrides) -> Params:
    """輔助函式：產生具備合法預設值的 Params 實例。"""
    defaults = {
        "current_age": 30,
        "retirement_age": 60,
        "life_expectancy": 80,
        "current_savings": 500_000.0,
        "monthly_investment": 10_000.0,
        "monthly_expense_today": 30_000.0,
        "annual_recurring_expense": 0.0,
        "labor_insurance_pension": 0.0,
        "labor_insurance_start_age": 0,
        "labor_pension_monthly": 0.0,
        "labor_pension_start_age": 0,
        "other_income": 0.0,
        "pre_retirement_return": 0.08,
        "post_retirement_return": 0.04,
        "inflation_rate": 0.02,
        "lump_sums": (),
    }
    defaults.update(overrides)
    return Params(**defaults)

def test_prd01_accumulation_zero_return():
    """PRD-01: r = 0 時退化為線性累加。"""
    ac, ar = 30, 60
    current_savings = 200_000.0
    monthly_investment = 15_000.0
    p = make_params(
        current_age=ac,
        retirement_age=ar,
        current_savings=current_savings,
        monthly_investment=monthly_investment,
        pre_retirement_return=0.0,
    )
    res = calculate(p)
    months = (ar - ac) * 12
    expected = current_savings + monthly_investment * months
    assert res.projected_savings == pytest.approx(expected, rel=1e-6)


def test_prd01_accumulation_compound_interest():
    """PRD-01: 統一採名目月利率 r/12 月複利，月存為期末年金。"""
    ac, ar = 30, 60
    r = 0.06
    r_m = r / 12.0
    n = (ar - ac) * 12
    s0 = 500_000.0
    pmt = 20_000.0

    fv_savings = s0 * ((1.0 + r_m) ** n)
    fv_annuity = pmt * (((1.0 + r_m) ** n - 1.0) / r_m)
    expected = fv_savings + fv_annuity

    p = make_params(
        current_age=ac,
        retirement_age=ar,
        current_savings=s0,
        monthly_investment=pmt,
        pre_retirement_return=r,
    )
    res = calculate(p)
    assert res.projected_savings == pytest.approx(expected, rel=1e-6)


def test_prd01_accumulation_zero_savings_and_investment():

def test_prd02_target_fund_annuity_due_formula():
    """PRD-02: 期初年金折現公式驗證 (第一期不折現)。"""
    ac, ar, ad = 35, 65, 80
    rp = 0.05
    monthly_exp = 30_000.0
    annual_exp = monthly_exp * 12.0
    n_years = ad - ar + 1

    # 期初年金現值 = sum_{k=0}^{n-1} CF / (1 + rp)^k
    expected_target = sum(annual_exp / ((1.0 + rp) ** k) for k in range(n_years))

    p = make_params(
        current_age=ac,
        retirement_age=ar,
        life_expectancy=ad,
        monthly_expense_today=monthly_exp,
        annual_recurring_expense=0.0,
        inflation_rate=0.0,
        post_retirement_return=rp,
        lump_sums=(),
    )
    res = calculate(p)
    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)


def test_prd02_inflation_locked_to_current_age():
    """PRD-02: 支出端通膨基準鎖現齡 A_c。"""
    ac, ar, ad = 30, 60, 65

    monthly_exp = 20_000.0

        monthly_expense_today=monthly_exp,
        annual_recurring_expense=0.0,
        inflation_rate=i,
        post_retirement_return=0.0,
        lump_sums=(),
    )
    res = calculate(p)
    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)

    """PRD-03 & PRD-12: 禁止 Math.max(0, ...) 截斷真實餘額，負值照實輸出。"""
    p = make_params(
        current_age=30,
        retirement_age=60,
        life_expectancy=80,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=100_000.0,
    )
    res = calculate(p)
    # 零資產高支出情況下，退休後餘額必然會穿破至負值
    assert any(b < 0 for b in res.balances_raw)
    assert any(b < 0 for b in res.balances_charted)
    # 圖表餘額不得截成 0，必須精確相等
    for raw, charted in zip(res.balances_raw, res.balances_charted):
        if raw < 0:
            assert charted == raw
            assert charted < 0

def test_prd05_lump_sum_outside_retirement_window_ignored():
    """PRD-05: A_e < A_r 或 A_e > A_d 之大筆支出皆排除不計入。"""

    assert res_pre.target_fund == pytest.approx(res_base.target_fund, rel=1e-6)

    # 壽命後支出 (A_e = 85 > 80)
    p_post = make_params(
        current_age=ac,

        lump_sums=(LumpSum(age=85, amount=1_000_000.0),),
    )
    res_post = calculate(p_post)
    assert res_post.target_fund == pytest.approx(res_base.target_fund, rel=1e-6)


def test_prd05_lump_sum_within_window_discounted_and_indexed():
    """PRD-05: 退休區間內的大筆支出鎖定 A_c 通膨後折現至 A_r。"""
    ac, ar, ad, ae = 30, 60, 80, 68
    i = 0.02
    rp = 0.04
    lump_amt = 800_000.0

    p_base = make_params(
        current_age=ac,
        retirement_age=ar,
        life_expectancy=ad,
        inflation_rate=i,
        post_retirement_return=rp,
        lump_sums=(),
    )
    p_lump = make_params(
        current_age=ac,
        retirement_age=ar,
        life_expectancy=ad,
        inflation_rate=i,
        post_retirement_return=rp,
        lump_sums=(LumpSum(age=ae, amount=lump_amt),),
    )

    res_base = calculate(p_base)
    res_lump = calculate(p_lump)

    # 名目金額: 金額 × (1+i)^(A_e - A_c)
    nominal_lump = lump_amt * ((1.0 + i) ** (ae - ac))
    # 折現至 A_r: 名目金額 / (1+r_p)^(A_e - A_r)
    discounted_lump = nominal_lump / ((1.0 + rp) ** (ae - ar))

    delta_target = res_lump.target_fund - res_base.target_fund
    assert delta_target == pytest.approx(discounted_lump, rel=1e-5)


def test_prd05_lump_sum_boundary_ages():
    """PRD-05: 恰好在 A_r 與 A_d 年的大筆支出皆須計入。"""
    ac, ar, ad = 30, 60, 80

def test_prd10_pension_indexing_locks_claim_age():
    """PRD-10: 勞保/勞退通膨基準鎖定請領年齡 A_p，第 A_p 年名目額為輸入金額*12。"""
    ac, ar, ad, ap = 30, 65, 65, 65
    i = 0.03
    pension_monthly = 20_000.0
    p = make_params(
        current_age=ac,
        retirement_age=ar,
        life_expectancy=ad,
        monthly_expense_today=50_000.0,
        annual_recurring_expense=0.0,
        labor_insurance_pension=pension_monthly,
        labor_insurance_start_age=ap,
        inflation_rate=i,
        post_retirement_return=0.0,
    )
    res = calculate(p)

    # 退休僅 65 歲這 1 年
    # 支出鎖 A_c: 50,000 * 12 * (1 + 0.03)^(65 - 30)
    exp = 50_000.0 * 12.0 * ((1.0 + i) ** (65 - ac))
    # 勞保鎖 A_p: 20,000 * 12 * (1 + 0.03)^(65 - 65) = 240,000
    inc = pension_monthly * 12.0 * ((1.0 + i) ** (65 - ap))
    expected_target = exp - inc

    assert res.target_fund == pytest.approx(expected_target, rel=1e-5)
