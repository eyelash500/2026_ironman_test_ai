# 1. 介面合規性與結構測試（Interface & Data Structure Tests）

def test_calculate_returns_result_instance(base_params: Params):
    """驗證 calculate 正確回傳 Result 型別實例。"""
    res = calculate(base_params)
    assert isinstance(res, Result)


def test_result_fields_types(base_params: Params):
    """驗證 Result 各欄位之資料型別符合型態定義。"""
    res = calculate(base_params)

    assert isinstance(res.projected_savings, (int, float))
    assert isinstance(res.target_fund, (int, float))
    assert isinstance(res.retirement_gap, (int, float))
    assert isinstance(res.balances_raw, tuple)
    assert isinstance(res.balances_charted, tuple)

    assert not math.isnan(res.projected_savings)
    assert not math.isnan(res.target_fund)
    assert not math.isnan(res.retirement_gap)
    assert not math.isinf(res.projected_savings)
    assert not math.isinf(res.target_fund)
    assert not math.isinf(res.retirement_gap)


def test_balance_series_elements_are_numeric(base_params: Params):
    """驗證 balances 序列中所有元素皆為數值且長度一致。"""
    res = calculate(base_params)

    assert len(res.balances_raw) > 0
    assert len(res.balances_raw) == len(res.balances_charted)
    assert all(isinstance(x, (int, float)) for x in res.balances_raw)
    assert all(isinstance(x, (int, float)) for x in res.balances_charted)
    assert all(not math.isnan(x) for x in res.balances_raw)
    assert all(not math.isnan(x) for x in res.balances_charted)

def test_frozen_dataclasses_immutability(base_params: Params):
    """驗證 LumpSum, Params, Result 皆宣告為 frozen，防止執行期任意改動。"""
    res = calculate(base_params)
    ls = LumpSum(age=50, amount=500_000.0)

    with pytest.raises(Exception):
        base_params.current_age = 35  # type: ignore

    with pytest.raises(Exception):
        res.projected_savings = 0.0  # type: ignore

    with pytest.raises(Exception):
        ls.amount = 100.0  # type: ignore

def test_monotonicity_higher_current_savings_yields_higher_projected_savings():
    """單調性：初始儲蓄較高者，退休時累積儲蓄必然較高。"""
    p_low = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=100_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=30_000.0,
    )
    p_high = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=1_000_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=30_000.0,
    )
    res_low = calculate(p_low)
    res_high = calculate(p_high)
    assert res_high.projected_savings > res_low.projected_savings


def test_monotonicity_higher_monthly_investment_yields_higher_projected_savings():
    """單調性：每月定期投資額較高者，退休時累積儲蓄必然較高。"""
    p_low = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=100_000.0,
        monthly_investment=5_000.0,
        monthly_expense_today=30_000.0,
    )
    p_high = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=100_000.0,
        monthly_investment=25_000.0,
        monthly_expense_today=30_000.0,
    )
    res_low = calculate(p_low)
    res_high = calculate(p_high)
    assert res_high.projected_savings > res_low.projected_savings


def test_monotonicity_higher_expense_yields_higher_target_fund():

def test_zero_savings_and_zero_investment():
    """邊界：完全無儲蓄且無每月投入時，計算應正常執行且 projected_savings >= 0。"""
    p = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=0.0,
        monthly_investment=0.0,
        monthly_expense_today=30_000.0,
    )
    res = calculate(p)
    assert isinstance(res, Result)
    assert res.projected_savings >= 0.0


def test_zero_rates_and_inflation():
    """邊界：報酬率與通膨率皆為 0% 時，計算核心不得拋出 ZeroDivisionError。"""
    p = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=200_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=30_000.0,
        pre_retirement_return=0.0,
        post_retirement_return=0.0,
        inflation_rate=0.0,
    )
    res = calculate(p)
    assert isinstance(res, Result)
    assert not math.isnan(res.projected_savings)
    assert not math.isnan(res.target_fund)


def test_immediate_retirement():
    """邊界：當前年齡即退休年齡（current_age == retirement_age），應正常計算。"""
    p = Params(
        current_age=65,
        retirement_age=65,
        life_expectancy=85,
        current_savings=10_000_000.0,
        monthly_investment=0.0,
        monthly_expense_today=50_000.0,
    )
    res = calculate(p)
    assert isinstance(res, Result)
    assert res.projected_savings >= 0.0
    assert res.target_fund > 0.0


def test_empty_and_multiple_lump_sums():
    """邊界：多筆單筆收支事件（包含正數與負數額度）應可正常輸入計算。"""
    p_none = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=500_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=30_000.0,
        lump_sums=(),
    )
    p_multi = Params(
        current_age=30,
        retirement_age=65,
        life_expectancy=85,
        current_savings=500_000.0,
        monthly_investment=10_000.0,
        monthly_expense_today=30_000.0,
        lump_sums=(
            LumpSum(age=35, amount=100_000.0),
            LumpSum(age=45, amount=-200_000.0),
            LumpSum(age=75, amount=50_000.0),
        ),
    )
    res_none = calculate(p_none)
    res_multi = calculate(p_multi)
    assert isinstance(res_none, Result)
    assert isinstance(res_multi, Result)
